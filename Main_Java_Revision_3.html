 77class Rect {
 78    private double len;
 79    private double brd;
 80
 81    public Rect() { }
 82    public Rect(double l, double b) {            // constructor
 83        len = l;
 84        brd = b;
 85    }
 86    public double getLen() { return len; }       // getters
 87    public double getBrd() { return brd; }
 88    public void setLen(double len) { this.len = len; }   // setters
 89    public void setBrd(double brd) { this.brd = brd; }
 90    public double area() { return len * brd; }
 91}

129class Complex {
130    private double re, im;
131    public Complex(double re, double im) { this.re = re; this.im = im; }
132    public Complex add(Complex o) {
133        return new Complex(this.re + o.re, this.im + o.im);
134    }

136    public String toString() { return re + " + " + im + "i"; }
