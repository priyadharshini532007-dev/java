/* =====================================================================
   1. GETTER / SETTER  +  CONSTRUCTOR  (Rectangle slides)
   ===================================================================== */
class Rect {
    private double len;
    private double brd;

    public Rect() { }
    public Rect(double l, double b) {            // constructor
        len = l;
        brd = b;
    }
    public double getLen() { return len; }       // getters
    public double getBrd() { return brd; }
    public void setLen(double len) { this.len = len; }   // setters
    public void setBrd(double brd) { this.brd = brd; }
    public double area() { return len * brd; }
}

/**
 * <h1>Find average of three numbers!</h1>
 * @author  Pratik Agarwal
 * @version 1.0
 * @since   2017-02-18
 */
class FindAvg {
    /**
     * @param a first parameter
     * @param b second parameter
     * @param c third parameter
     * @return int average of a, b and c
     */
    public int findAvg(int a, int b, int c) {
        return (a + b + c) / 3;
    }
}


/* =====================================================================
   4. NO OPERATOR OVERLOADING / FRIEND FUNCTION -> add() method
   ===================================================================== */
class Complex {
    private double re, im;
    public Complex(double re, double im) { this.re = re; this.im = im; }
    public Complex add(Complex o) {
        return new Complex(this.re + o.re, this.im + o.im);
    }
    @Override
    public String toString() { return re + " + " + im + "i"; }
}


/* =====================================================================
   5. INHERITANCE  (Vehicle -> Motorcar)  only public inheritance in Java
   ===================================================================== */
class Vehicle {
    private String color;
    private int speed, seats, wheels;
    public Vehicle(String c, int s, int n1, int n2) {
        color = c; speed = s; seats = n1; wheels = n2;
    }
    public void display() {
        System.out.println("Color: " + color);
        System.out.println("Max Speed: " + speed);
        System.out.println("Num Seats: " + seats);
        System.out.println("Num Wheels: " + wheels);
    }
}

class Motorcar extends Vehicle {
    private int doors;
    public Motorcar(String c, int s, int n1, int n2, int n3) {
        super(c, s, n1, n2);
        doors = n3;
    }
    @Override
    public void display() {
        super.display();
        System.out.println("Number of Doors: " + doors);
    }
}

/* =====================================================================
   7. INTERFACES
      methods -> public abstract ; fields -> public static final
      interface cannot be instantiated
      class implements interface | interface extends interface | class extends class
   ===================================================================== */
interface Area {
    final static float pi = 3.14F;
    float compute(float x, float y);
}
class RectArea implements Area {
    public float compute(float x, float y) { return (x * y); }
}
class Circle implements Area {
    public float compute(float x, float y) { return (pi * x * x); }
}

// ---- hybrid: extends class + implements interface ----
class Student {
    int rollNumber;
    void getNumber(int n) { rollNumber = n; }
    void putNumber() { System.out.println("Roll No : " + rollNumber); }
}
class Test extends Student {
    float part1, part2;
    void getMarks(float m1, float m2) { part1 = m1; part2 = m2; }
    void putMarks() {
        System.out.println("Marks obtained ");
        System.out.println("part 1 = " + part1);
        System.out.println("Part2 = " + part2);
    }
}
interface Sports {
    float sportWt = 6.0F;
    void putwt();
}
class Results extends Test implements Sports {
    float total;
    public void putwt() { System.out.println("Sports Wt = " + sportWt); }
    void display() {
        total = part1 + part2 + sportWt;
        putNumber();
        putMarks();
        putwt();
        System.out.println("Total score = " + total);
    }
}

// ---- interface extends interface ----
interface ParentInterface { void displayParent(); }
interface ChildInterface extends ParentInterface { void displayChild(); }
class Both implements ChildInterface {           // must implement BOTH
    public void displayParent() { System.out.println("Parent method implementation"); }
    public void displayChild()  { System.out.println("Child method implementation"); }
}

// ---- diamond (variables) ----
interface A { int VAL_A = 10; void display(); }
interface B extends A { int VAL_B = 20; void showB(); }
interface C extends A { int VAL_C = 30; void showC(); }
class D implements B, C {
    public void display() { System.out.println("VAL_A = " + VAL_A); }
    public void showB()   { System.out.println("VAL_B = " + VAL_B); }
    public void showC()   { System.out.println("VAL_C = " + VAL_C); }
}

      Types: 1. Non-static (member)  2. Static nested  3. Local  4. Anonymous

class DataStructure {
    private int[] data = {1, 2, 3, 4, 5};

    class Iterator {                             // non-static inner class
        int index = 0;
        boolean hasNext() { return index < data.length; }
        int next() { return data[index++]; }     // direct access to private data
    }
    public Iterator getIterator() { return new Iterator(); }
}

class HandledDemo {
    public static void method2() {
        System.out.println("--- Inside method2 ---");
        int a = 10;
        int b = 0;
        try {
            int result = a / b;
            System.out.println("Result: " + result);
        } catch (ArithmeticException e) {
            System.out.println("ERROR: Division by zero caught gracefully.");
        }
        System.out.println("method2 finished.");
    }
}

class UnhandledDemo {
    public static void method1() {
        System.out.println("--- Inside method1 ---");
        int a = 10;
        int b = 0;
        int result = a / b;                      // program terminates here
        System.out.println("Result: " + result); // never runs
        System.out.println("method1 finished.");
    }
}
